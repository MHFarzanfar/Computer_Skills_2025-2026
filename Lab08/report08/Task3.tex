\providecommand{\Square}[2]{\fill #1 rectangle ++(#2,#2);}
\begin{tikzpicture}
  \tikzmath{
    function carpet(\x, \y, \s, \d) {
      if (\d == 0) then {
        { \Square{(\x,\y)}{\s}; };
      } else {
        let \u = \s / 3;
        for \i in {0,1,2}{
          for \j in {0,1,2}{
            if !(\i==1 && \j==1) then {
              carpet(\x + \i*\u, \y + \j*\u, \u, \d-1);
            };
          };
        };
      };
    };
    carpet(0, 0, 9, 3);
  }
\end{tikzpicture}
